3 ОПИС ВИКОНАНОЇ РОБОТИ
Під час виконання практичного завдання було детально опрацьовано тему методів рефакторингу програмного коду на прикладах, реалізованих мовою F#. Робота розпочалася зі створення титульного слайду та визначення основної мети дослідження - покращення структури та якості програмного коду шляхом застосування різних технік рефакторингу, що стало основою подальшої практичної частини (рис. Б.1).
Далі було розглянуто загальне поняття рефакторингу, його призначення та ключові принципи. У цій частині було підкреслено, що рефакторинг дозволяє оптимізувати внутрішню структуру програми без зміни її функціональності, робить код зрозумілішим, легшим у підтримці (рис. Б.2).
Наступним кроком було визначення та опис трьох основних методів, які аналізувалися в роботі: Extract Method, Rename Method та Simplify Conditional Expressions. У цьому блоці було окреслено суть кожного методу, його призначення та ключові переваги, що дозволяє зрозуміти, чому саме ці техніки широко застосовуються під час розробки програмного забезпечення (рис. Б.3).
Після цього було проведено детальний аналіз прикладу для методу Extract Method. Було розглянуто великий фрагмент коду, у якому логіка обчислення середніх значень, визначення статусу та бонусів була змішана в одному методі. Виявлені проблеми включали дублювання логіки, надмірне використання змінних умовного типу mutable та складність тестування (рис. Б.4). Після рефакторингу код було поділено на окремі функції, що значно покращило структуру та читабельність (рис. Б.5).
Також було розглянуто приклад методу Rename Method. У початковому варіанті коду використовувалися короткі, неінформативні назви змінних, через що логіку обчислення було важко зрозуміти (рис. Б.6). Після застосування методу відбулося суттєве підвищення зрозумілості коду завдяки перейменуванню змінних та функцій на такі, що чітко відображають їх призначення. (рис. Б.7).
Метод Simplify Conditional Expressions був продемонстрований на прикладі складного каскаду умовних операторів. У вихідному варіанті код був громіздким та важким для читання (рис. Б.8). Після рефакторингу умовні конструкції були замінені більш читабельною структурою match…with, що дозволило зробити логіку компактною, наочною та простою для розширення. (рис. Б.9).
Окремо був розглянутий блок про інструменти для рефакторингу. Було описано найпоширеніші середовища розробки та аналізатори коду, які полегшують процес рефакторингу: Visual Studio, JetBrains Rider, Eclipse, а також інструменти статичного аналізу. Для кожного інструмента було наведено його ключові можливості, корисні під час рефакторингу, що допомагає значно покращити якість коду в реальних проєктах (рис. Б.10).
Наприкінці роботи було сформовано загальні висновки щодо важливості регулярного рефакторингу та його впливу на якість коду. Було зазначено, що рефакторинг слід проводити систематично, а не лише тоді, коли код стає складним у підтримці. Також були визначені переваги застосування базових методів рефакторингу та описано, як вони покращують розуміння всього проєкту (рис. Б.11).

  3
  4 let generateCourseReport (students: Student list) =
  5     let mutable report = []
  6     for s in students do
  7         let scoreSum = List.sum s.Scores
  8         let hwSum = List.sum s.Homework
  9         let avgScore = (scoreSum + hwSum) / (List.length s.Scores + List.length s.Homework)
 10
 11         let mutable status = ""

 13             status <- "Excellent"

 17             status <- "Satisfactory"

 19             status <- "Poor"
 20

 24

  1 //Гарний код
  2 type Student = { Name : string; Scores : int list; Attendance : int; Homework : int list }
  3
  4 let calculateAverage s =
  5     let scoreSum = List.sum s.Scores
  6     let hwSum = List.sum s.Homework
  7     (scoreSum + hwSum) / (s.Scores.Length + s.Homework.Length)
  8
  9 let determineStatus avg attendance =
 10     match avg, attendance with
 11     | a, att when a > 90 && att > 95 -> "Excellent"
 12     | a, att when a > 75 && att > 80 -> "Good"
 13     | a, att when a > 60 && att > 70 -> "Satisfactory"
 14     | _ -> "Poor"
 15
 16 let calculateBonus avg attendance =
 17     let bonusAttendance = if attendance > 98 then 5 else 0
 18     let bonusScore = if avg > 95 then 10 else 0
 19     bonusAttendance + bonusScore
 20
 21 let generateCourseReport students =
 22     students
 23     |> List.map (fun s ->
 24         let avg = calculateAverage s
 25         let status = determineStatus avg s.Attendance
 26         let bonus = calculateBonus avg s.Attendance
 27         let finalScore = avg + bonus
 28         (s.Name, finalScore, status))
